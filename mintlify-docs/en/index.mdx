export const Motes = ({ rgb, count }) => {
  const ref = useRef(null);
  useEffect(() => {
    const c = ref.current;
    if (!c) return;
    const ctx = c.getContext("2d");
    let w = 0, h = 0, raf = 0;
    const dpr = Math.min(window.devicePixelRatio || 1, 2);
    const resize = () => { w = c.clientWidth; h = c.clientHeight; c.width = w * dpr; c.height = h * dpr; ctx.setTransform(dpr, 0, 0, dpr, 0, 0); };
    resize();
    const ro = new ResizeObserver(resize);
    ro.observe(c);
    const ps = Array.from({ length: count || 90 }, () => ({ x: Math.random(), y: Math.random(), r: Math.random() * 1.7 + 0.3, v: Math.random() * 0.0004 + 0.00008, p: Math.random() * 6.28 }));
    const draw = (t) => {
      ctx.clearRect(0, 0, w, h);
      for (const s of ps) {
        s.y -= s.v;
        if (s.y < 0) { s.y = 1; s.x = Math.random(); }
        const a = 0.25 + 0.5 * Math.sin(t / 700 + s.p);
        ctx.beginPath();
        ctx.arc((s.x + Math.sin(t / 2400 + s.p) * 0.012) * w, s.y * h, s.r, 0, 6.283);
        ctx.fillStyle = "rgba(" + rgb + "," + Math.max(a, 0.06) + ")";
        ctx.fill();
      }
      raf = requestAnimationFrame(draw);
    };
    raf = requestAnimationFrame(draw);
    return () => { cancelAnimationFrame(raf); ro.disconnect(); };
  }, []);
  return <canvas ref={ref} style={{ position: "absolute", inset: 0, width: "100%", height: "100%", pointerEvents: "none" }} />;
};

export const Parallax = () => {
  const ref = useRef(null);
  useEffect(() => {
    const el = ref.current && ref.current.parentElement;
    if (!el) return;
    const mv = (e) => {
      const r = el.getBoundingClientRect();
      el.style.setProperty("--mx", ((e.clientX - r.left) / r.width - 0.5).toFixed(3));
      el.style.setProperty("--my", ((e.clientY - r.top) / r.height - 0.5).toFixed(3));
    };
    el.addEventListener("mousemove", mv);
    return () => el.removeEventListener("mousemove", mv);
  }, []);
  return <span ref={ref} aria-hidden="true" style={{ display: "none" }} />;
};
